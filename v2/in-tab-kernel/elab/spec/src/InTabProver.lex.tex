definition{
  name: proverVerdict,
  parameters: [e : InTabKernel::Evidence, r : InTabElaborator::Run],
  result: Nat,
  body: (if (((project e leanRan) =? 1) && (not (InTabElaborator::accepted r))) then (InTabKernel::codeRefused) else (InTabKernel::verdictCode e))
}

definition{
  name: breaksKernel,
  parameters: [e : InTabKernel::Evidence, r : InTabElaborator::Run],
  result: Bool,
  body: ((InTabKernel::accepted (proverVerdict e r)) && (not (InTabKernel::accepted (InTabKernel::verdictCode e))))
}

  name: nothingTheKernelRefusesIsAccepted,

  statement: ((breaksKernel e r) = false),
  proof: linear_arithmetic{
    definitions: [
      InTabKernel::accepted,

      breaksKernel,

  },
  axioms: [Classical.choice, Quot.sound, propext]
}
