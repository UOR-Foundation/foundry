definition{
  name: hasEvidence,
  parameters: [evidence : (List String)],
  result: Bool,
  body: (match evidence | List.nil => false | List.cons item others => true)
}

definition{
  name: obligationsEvidenced,
  parameters: [position : Nat, items : (List Obligation)],
  result: Bool,
  body: (match items
    | List.nil => true
    | List.cons item rest => (((project item index) =? position) && ((hasEvidence (project item evidence)) && (obligationsEvidenced (position + 1) rest)))),
  recursive_argument: items
}
