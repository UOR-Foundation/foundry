  name: WriteRequest,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: granted, type: Nat},

    {name: existing, type: Nat},

definition{
  name: writeAllows,
  parameters: [request : WriteRequest],
  result: Nat,
  body: (if ((project request denied) =? 1) then 0 else (if (((project request granted) =? 1) && ((project request rootMatch) =? 1)) then (if ((project request existing) =? 1) then 1 else (if ((project request quota) <? ((project request used) + (project request size))) then 0 else 1)) else 0))
}

definition{
  name: writeCharge,
  parameters: [request : WriteRequest],
  result: Nat,
  body: (if ((project request existing) =? 1) then 0 else (project request size))
}

definition{
  name: breaksWriteWithinQuota,
  parameters: [request : WriteRequest],
  result: Bool,
  body: (((not ((project request existing) =? 1)) && ((project request quota) <? ((project request used) + (project request size)))) && ((writeAllows request) =? 1))
}

theorem{
  name: writeWithinQuotaHolds,
  parameters: [request : WriteRequest],

  axioms: [Quot.sound, propext]
}

  name: breaksWriteNeedsGrantAndRoot,

}

theorem{
  name: writeNeedsGrantAndRootHolds,
