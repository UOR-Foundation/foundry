  name: childHolds,
  parameters: [request : DelegationRequest],

  name: childQuota,
  parameters: [request : DelegationRequest],
  result: Nat,
  body: (if ((project request parentQuota) <? (project request childAsk)) then (project request parentQuota) else (project request childAsk))
}

definition{
  name: prepareStarts,
  parameters: [request : DelegationRequest],
  result: Nat,
  body: (if ((((project request parentAdmitted) =? 1) && ((project request delegationAdmitted) =? 1)) && ((project request childAdmitted) =? 1)) then 1 else 0)
}

  name: breaksChildNeverExceedsParent,
  parameters: [request : DelegationRequest],

  name: breaksChildNeedsDelegation,
